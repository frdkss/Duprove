cmake_minimum_required(VERSION 3.24)
project(duprove LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
FetchContent_Declare(sha GIT_REPOSITORY https://github.com/okdshin/PicoSHA2.git GIT_TAG 161cb3fc4170fa7a3eca9e582cebd27cc4d1fe29)
FetchContent_Declare(sqlite URL https://www.sqlite.org/2025/sqlite-amalgamation-3500400.zip)
FetchContent_MakeAvailable(json sha sqlite)
add_library(sqlite3 STATIC ${sqlite_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite3 PUBLIC ${sqlite_SOURCE_DIR})
add_library(duprove-core STATIC src/platform.cpp src/mounts.cpp src/scanner.cpp src/context.cpp src/comparison.cpp src/traversal.cpp src/report.cpp)
target_include_directories(duprove-core PUBLIC src PRIVATE ${sha_SOURCE_DIR})
target_link_libraries(duprove-core PUBLIC sqlite3 nlohmann_json::nlohmann_json)
add_executable(duprove-analyzer src/main.cpp)
target_link_libraries(duprove-analyzer PRIVATE duprove-core)
if(WIN32)
  target_compile_definitions(duprove-core PUBLIC _WIN32_WINNT=0x0601 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(duprove-core PUBLIC shell32 ole32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(duprove-core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
endif()
foreach(target duprove-core duprove-analyzer)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(duprove-core-test tests/core_test.cpp)
  target_link_libraries(duprove-core-test PRIVATE duprove-core)
  if(MSVC)
    target_compile_options(duprove-core-test PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME core-contract COMMAND duprove-core-test)
  add_executable(duprove-mounts-test tests/mounts_test.cpp)
  target_link_libraries(duprove-mounts-test PRIVATE duprove-core)
  add_test(NAME mount-discovery COMMAND duprove-mounts-test)
endif()
set_target_properties(duprove-analyzer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER ${config} upper)
  set_target_properties(duprove-analyzer PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
