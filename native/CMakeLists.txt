cmake_minimum_required(VERSION 3.24)
project(duprove LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json
  URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_Declare(sha GIT_REPOSITORY https://github.com/okdshin/PicoSHA2.git GIT_TAG 161cb3fc4170fa7a3eca9e582cebd27cc4d1fe29)
FetchContent_Declare(sqlite
  URL https://www.sqlite.org/2025/sqlite-amalgamation-3500400.zip
  URL_HASH SHA256=1d3049dd0f830a025a53105fc79fd2ab9431aea99e137809d064d8ee8356b032)
FetchContent_MakeAvailable(json sha sqlite)
include(cmake/content-dependencies.cmake)
add_library(sqlite3 STATIC ${sqlite_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite3 PUBLIC ${sqlite_SOURCE_DIR})
add_library(duprove-core STATIC src/platform.cpp src/mounts.cpp src/file_access.cpp src/user_dirs.cpp src/scanner.cpp src/context.cpp src/comparison.cpp src/traversal.cpp src/report.cpp)
target_sources(duprove-core PRIVATE src/content.cpp src/content_text.cpp src/content_docx.cpp src/content_image.cpp src/parallel_hash.cpp src/local_file.cpp src/recycle.cpp)
if(APPLE)
  enable_language(OBJCXX)
  set_target_properties(duprove-core PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
  target_sources(duprove-core PRIVATE src/user_dirs_mac.mm)
  target_link_libraries(duprove-core PUBLIC "-framework Foundation")
endif()
target_include_directories(duprove-core PUBLIC src PRIVATE ${sha_SOURCE_DIR})
target_link_libraries(duprove-core PUBLIC sqlite3 nlohmann_json::nlohmann_json PRIVATE duprove-content-libraries)
add_executable(duprove-analyzer src/main.cpp)
target_link_libraries(duprove-analyzer PRIVATE duprove-core)
if(WIN32)
  target_compile_definitions(duprove-core PUBLIC _WIN32_WINNT=0x0601 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(duprove-core PUBLIC shell32 ole32 bcrypt)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(duprove-core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
endif()
foreach(target duprove-core duprove-analyzer)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(duprove-core-test tests/core_test.cpp)
  target_link_libraries(duprove-core-test PRIVATE duprove-core)
  if(MSVC)
    target_compile_options(duprove-core-test PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME core-contract COMMAND duprove-core-test)
  add_executable(duprove-report-test tests/report_test.cpp)
  target_link_libraries(duprove-report-test PRIVATE duprove-core)
  add_test(NAME anonymous-statistics COMMAND duprove-report-test)
  add_executable(duprove-mounts-test tests/mounts_test.cpp)
  target_link_libraries(duprove-mounts-test PRIVATE duprove-core)
  add_test(NAME mount-discovery COMMAND duprove-mounts-test)
  add_executable(duprove-traversal-test tests/traversal_test.cpp)
  target_link_libraries(duprove-traversal-test PRIVATE duprove-core)
  if(MSVC)
    target_compile_options(duprove-traversal-test PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME safe-traversal COMMAND duprove-traversal-test)
  add_executable(duprove-comparison-test tests/comparison_test.cpp)
  target_link_libraries(duprove-comparison-test PRIVATE duprove-core)
  if(MSVC)
    target_compile_options(duprove-comparison-test PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME exact-comparison COMMAND duprove-comparison-test)
  add_executable(duprove-parallel-hash-test tests/parallel_hash_test.cpp)
  target_link_libraries(duprove-parallel-hash-test PRIVATE duprove-core)
  add_test(NAME parallel-hashing COMMAND duprove-parallel-hash-test)
  set_tests_properties(parallel-hashing PROPERTIES TIMEOUT 30)
  add_executable(duprove-resource-budget-test tests/resource_budget_test.cpp)
  target_link_libraries(duprove-resource-budget-test PRIVATE duprove-core)
  add_test(NAME resource-budgets COMMAND duprove-resource-budget-test)
  set_tests_properties(resource-budgets PROPERTIES TIMEOUT 30)
  add_executable(duprove-content-test tests/content_test.cpp)
  target_link_libraries(duprove-content-test PRIVATE duprove-core duprove-content-libraries)
  if(MSVC)
    target_compile_options(duprove-content-test PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME content-comparison COMMAND duprove-content-test)
endif()
set_target_properties(duprove-analyzer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER ${config} upper)
  set_target_properties(duprove-analyzer PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
