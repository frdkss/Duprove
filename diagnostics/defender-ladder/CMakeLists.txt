cmake_minimum_required(VERSION 3.24)
project(duprove_defender_ladder LANGUAGES C CXX)
if(NOT MSVC OR NOT WIN32)
  message(FATAL_ERROR "This diagnostic compares Windows MSVC binaries only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "Do not build production tests" FORCE)
get_filename_component(DUPROVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Reuse the already downloaded sources, but compile in this isolated build tree.
foreach(dependency json sha sqlite miniz pugixml cmark lodepng turbojpeg)
  string(TOUPPER "${dependency}" upper)
  set(source "${DUPROVE_ROOT}/build/native/_deps/${dependency}-src")
  if(EXISTS "${source}")
    set("FETCHCONTENT_SOURCE_DIR_${upper}" "${source}" CACHE PATH "Existing dependency source")
  endif()
endforeach()
add_subdirectory("${DUPROVE_ROOT}/native" native EXCLUDE_FROM_ALL)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/maps")
set(labels
  "Minimal MSVC main, dynamic CRT"
  "Minimal MSVC main, static CRT"
  "B + SQLite entry points"
  "C + production Windows SHA256 (BCrypt/CNG)"
  "D + miniz ZIP reader entry points"
  "E + pugixml and cmark entry points"
  "F + lodepng and turbojpeg decoder entry points"
  "G + platform, mounts, user-directory support"
  "H + file access"
  "I + traversal and its context dependencies"
  "J + comparison and parallel hashing"
  "K + content analysis entry points")
set(entries "")
set(stage 0)
foreach(letter A B C D E F G H I J K L)
  math(EXPR stage "${stage}+1")
  math(EXPR index "${stage}-1")
  list(GET labels ${index} label)
  set(target "defender-${letter}")
  add_executable(${target} probes.cpp)
  target_compile_definitions(${target} PRIVATE DIAGNOSTIC_STAGE=${stage} NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  target_compile_options(${target} PRIVATE /utf-8 /W4)
  target_include_directories(${target} PRIVATE "${DUPROVE_ROOT}/native/src")
  set(runtime "MT")
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(stage EQUAL 1)
    set(runtime "MD")
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  endif()
  if(stage GREATER_EQUAL 3)
    target_link_libraries(${target} PRIVATE sqlite3)
  endif()
  if(stage GREATER_EQUAL 4)
    target_link_libraries(${target} PRIVATE bcrypt)
  endif()
  if(stage GREATER_EQUAL 5)
    target_link_libraries(${target} PRIVATE miniz)
  endif()
  if(stage GREATER_EQUAL 6)
    target_link_libraries(${target} PRIVATE pugixml::pugixml cmark)
  endif()
  if(stage GREATER_EQUAL 7)
    target_link_libraries(${target} PRIVATE duprove-png duprove-jpeg)
  endif()
  if(stage GREATER_EQUAL 8)
    target_link_libraries(${target} PRIVATE duprove-core)
  endif()
  target_link_options(${target} PRIVATE "/MAP:${CMAKE_BINARY_DIR}/maps/${letter}.map")
  set_property(TARGET ${target} PROPERTY RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
  string(APPEND entries "{\"stage\":\"${letter}\",\"label\":\"${label}\",\"runtime\":\"${runtime}\",\"file\":\"$<TARGET_FILE:${target}>\",\"map\":\"${CMAKE_BINARY_DIR}/maps/${letter}.map\"},\n")
endforeach()
target_link_options(duprove-analyzer PRIVATE "/MAP:${CMAKE_BINARY_DIR}/maps/M.map")
add_custom_target(defender-M DEPENDS duprove-analyzer)
string(APPEND entries "{\"stage\":\"M\",\"label\":\"Full analyzer compiled from unchanged production sources\",\"runtime\":\"MT\",\"file\":\"$<TARGET_FILE:duprove-analyzer>\",\"map\":\"${CMAKE_BINARY_DIR}/maps/M.map\"}")
set(control 0)
foreach(letter N O)
  math(EXPR control "${control}+1")
  set(target "defender-${letter}")
  add_executable(${target} controls.cpp)
  target_compile_definitions(${target} PRIVATE DIAGNOSTIC_CONTROL=${control} NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  target_compile_options(${target} PRIVATE /utf-8 /W4)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(${target} PRIVATE duprove-core miniz pugixml::pugixml cmark duprove-png duprove-jpeg)
  target_link_options(${target} PRIVATE "/MAP:${CMAKE_BINARY_DIR}/maps/${letter}.map")
  set_property(TARGET ${target} PROPERTY RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
  if(control EQUAL 1)
    set(label "L + report builder entry point, no analyzer CLI")
  else()
    set(label "N + scan entry point, no analyzer CLI")
  endif()
  string(APPEND entries ",\n{\"stage\":\"${letter}\",\"label\":\"${label}\",\"runtime\":\"MT\",\"file\":\"$<TARGET_FILE:${target}>\",\"map\":\"${CMAKE_BINARY_DIR}/maps/${letter}.map\"}")
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/manifest-$<CONFIG>.json" CONTENT
  "{\"compiler\":\"${CMAKE_CXX_COMPILER}\",\"compilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"generator\":\"${CMAKE_GENERATOR}\",\"dumpbin\":\"${CMAKE_DUMPBIN}\",\"configuration\":\"$<CONFIG>\",\"entries\":[${entries}]}\n")
